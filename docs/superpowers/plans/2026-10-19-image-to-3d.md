/** Hugging Face token của người dùng: chỉ lưu trong trình duyệt, chỉ gửi tới Hugging Face. */
export const TOKEN_KEY = "hand3d.hfToken";

export const isValidToken = (t: string) => /^hf_[A-Za-z0-9]{30,}$/.test(t.trim());

export function loadToken(): string {
  try {
    return localStorage.getItem(TOKEN_KEY) ?? "";
  } catch {
    return ""; // ngoài trình duyệt hoặc bị chặn lưu trữ
  }
}

export function saveToken(t: string): void {
  try {
    localStorage.setItem(TOKEN_KEY, t.trim());
  } catch {}
}

export function clearToken(): void {
  try {
    localStorage.removeItem(TOKEN_KEY);
  } catch {}
}

export const maskToken = (t: string) => `hf_••••${t.slice(-4)}`;
```

import { GenError } from "./genErrors.ts";

export const MAX_BYTES = 10 * 1024 * 1024;
export const MAX_SIDE = 1024;
export const ACCEPTED = ["image/jpeg", "image/png", "image/webp"];

export function checkImage(f: { type: string; size: number }): GenError | null {
  if (!ACCEPTED.includes(f.type)) return new GenError("bad_image", `Định dạng ${f.type || "không rõ"} không được hỗ trợ`);
  if (f.size > MAX_BYTES) return new GenError("bad_image", "Ảnh lớn hơn 10 MB");
  return null;
}

export function fitWithin(width: number, height: number, max = MAX_SIDE) {
  const k = Math.min(1, max / Math.max(width, height));
  return { width: Math.round(width * k), height: Math.round(height * k) };
}

/** Trình duyệt: kiểm tra rồi thu nhỏ ảnh về cạnh dài ≤ MAX_SIDE, xuất PNG. */
export async function prepImage(file: File): Promise<Blob> {
  const err = checkImage(file);
  if (err) throw err;
  let bmp: ImageBitmap;
  try {
    bmp = await createImageBitmap(file);
  } catch {
    throw new GenError("bad_image", "Không đọc được ảnh");
  }
  const { width, height } = fitWithin(bmp.width, bmp.height);
  const canvas = document.createElement("canvas");
  canvas.width = width;
  canvas.height = height;
  canvas.getContext("2d")?.drawImage(bmp, 0, 0, width, height);
  bmp.close();
  return await new Promise<Blob>((resolve, reject) =>
    canvas.toBlob((b) => (b ? resolve(b) : reject(new GenError("bad_image", "Không xử lý được ảnh"))), "image/png"),
  );
}
```

```ts
type StatusLike = { title?: unknown; message?: unknown };

function textOf(err: unknown): string {
  if (typeof err === "string") return err;
  if (err && typeof err === "object") {
    const { title, message } = err as StatusLike;
    const msg =
      typeof message === "string"
        ? message
        : Array.isArray(message)
          ? message.map((m) => (m && typeof m === "object" && "message" in m ? String(m.message) : String(m))).join("; ")
          : "";
    return [typeof title === "string" ? title : "", msg].filter(Boolean).join(": ");
  }
  return String(err);
}

export function classifyError(err: unknown, fallback: GenErrorCode = "unknown"): GenError {
  if (err instanceof GenError) return err;
  const text = textOf(err);
  if (/exceeded your ZeroGPU quota|ZeroGPU quota exceeded/i.test(text)) {
    const retryIn = text.match(/Try again in ([\d:]+)/)?.[1];
    // HF trả câu này khi request bị coi là ẩn danh → token thiếu hoặc không được chấp nhận
    const code = /Authenticate with a Hugging Face token/i.test(text) ? "bad_token" : "quota";
    return new GenError(code, text, retryIn);
  }
  if (/invalid (user )?token|unauthori[sz]ed|\b401\b|\b403\b/i.test(text)) return new GenError("bad_token", text);
  if (/ZeroGPU worker error|could not resolve app config|failed to fetch|networkerror|connection (error|closed)|space .*(sleeping|paused|building|runtime error)/i.test(text)) {
    return new GenError("space_down", text);
  }
  return new GenError(fallback, text);
}

export function errorMessage(e: GenError): string {
  switch (e.code) {
    case "bad_image":
      return `Ảnh không dùng được (${e.message}). Chọn file jpg, png hoặc webp dưới 10 MB.`;
    case "bad_token":
      return "Hugging Face không chấp nhận token này nên coi bạn là người dùng ẩn danh. Kiểm tra lại hoặc tạo token loại Read mới tại huggingface.co/settings/tokens.";
    case "quota":
      return `Đã hết lượt GPU miễn phí của tài khoản Hugging Face${e.retryIn ? `; thử lại sau ${e.retryIn}` : ""}.`;
    case "space_down":
      return "Space tạo 3D đang bảo trì hoặc quá tải. Thử lại sau ít phút.";
    case "timeout":
      return "Quá 6 phút vẫn chưa xong, hàng đợi đang đông. Thử lại sau.";
    case "cancelled":
      return "Đã huỷ.";
    default:
      return `Có lỗi: ${e.message || "không rõ"}. Thử lại.`;
  }
}
```

/** Adapter cho Space microsoft/TRELLIS.2: ảnh → file .glb. Client và fetch tiêm vào được để test. */
import { Client, handle_file } from "@gradio/client";
import { classifyError, GenError } from "./genErrors.ts";

export const SPACE_ID = "microsoft/TRELLIS.2";
export const TIMEOUT_MS = 6 * 60 * 1000;

export type Phase = "connecting" | "queued" | "preprocessing" | "generating" | "exporting" | "downloading";
export type PhaseInfo = { phase: Phase; position?: number };
export type SpaceEvent = { type: string; stage?: string; position?: number; title?: string; message?: unknown; data?: unknown };
export type SpaceJob = AsyncIterable<SpaceEvent> & { cancel: () => Promise<void> };
export type SpaceClient = { submit: (endpoint: string, data: unknown[] | Record<string, unknown>) => SpaceJob };
export type GenerateOptions = {
  onPhase?: (p: PhaseInfo) => void;
  signal?: AbortSignal;
  timeoutMs?: number;
  connect?: (token: string) => Promise<SpaceClient>;
  fetchFn?: typeof fetch;
};

const defaultConnect = async (token: string) =>
  (await Client.connect(SPACE_ID, { hf_token: token as `hf_${string}`, events: ["data", "status"] })) as unknown as SpaceClient;

export async function generate(image: Blob, token: string, opts: GenerateOptions = {}): Promise<Blob> {
  const { onPhase = () => {}, signal, timeoutMs = TIMEOUT_MS, connect = defaultConnect, fetchFn = fetch } = opts;
  const ctrl = new AbortController();
  const stopError = () => new GenError(ctrl.signal.reason === "timeout" ? "timeout" : "cancelled", "");
  // Promise bị reject khi huỷ/quá giờ — chạy đua với pipeline để không bao giờ treo
  const stopped = new Promise<never>((_, reject) => ctrl.signal.addEventListener("abort", () => reject(stopError())));
  const timer = setTimeout(() => ctrl.abort("timeout"), timeoutMs);
  const onAbort = () => ctrl.abort("cancelled");
  if (signal?.aborted) onAbort();
  signal?.addEventListener("abort", onAbort);

  const call = async (client: SpaceClient, endpoint: string, data: unknown[] | Record<string, unknown>, phase: Phase) => {
    if (ctrl.signal.aborted) throw stopError();
    onPhase({ phase });
    const job = client.submit(endpoint, data);
    const cancelJob = () => void job.cancel();
    ctrl.signal.addEventListener("abort", cancelJob);
    try {
      for await (const ev of job) {
        if (ev.type === "status") {
          if (ev.stage === "error") throw classifyError(ev);
          if (ev.stage === "pending" && ev.position) onPhase({ phase: "queued", position: ev.position });
          if (ev.stage === "generating") onPhase({ phase });
        } else if (ev.type === "data") {
          return (ev.data ?? []) as unknown[];
        }
      }
      throw new GenError("space_down", `${endpoint} kết thúc mà không trả dữ liệu`);
    } finally {
      ctrl.signal.removeEventListener("abort", cancelJob);
    }
  };

  const run = async () => {
    onPhase({ phase: "connecting" });
    const client = await connect(token).catch((e) => Promise.reject(classifyError(e, "space_down")));
    await call(client, "/start_session", [], "connecting");
    const [pre] = await call(client, "/preprocess_image", [handle_file(image)], "preprocessing");
    await call(client, "/image_to_3d", { image: pre, resolution: "1024" }, "generating");
    const out = await call(client, "/extract_glb", { decimation_target: 100000, texture_size: 1024 }, "exporting");
    const url = (out as ({ url?: string } | null)[]).find((f) => f?.url)?.url;
    if (!url) throw new GenError("space_down", "Space không trả về file .glb");
    onPhase({ phase: "downloading" });
    const res = await fetchFn(url, { signal: ctrl.signal });
    if (!res.ok) throw new GenError("space_down", `Tải file .glb lỗi HTTP ${res.status}`);
    return await res.blob();
  };

  try {
    return await Promise.race([run(), stopped]);
  } catch (e) {
    throw ctrl.signal.aborted ? stopError() : classifyError(e);
  } finally {
    clearTimeout(timer);
    signal?.removeEventListener("abort", onAbort);
  }
}
